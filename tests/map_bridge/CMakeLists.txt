cmake_minimum_required(VERSION 3.20)
project(map_bridge_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
# Resolve the maintained crate from this standalone project's repository root.
set(CMAKE_SOURCE_DIR "${REPO}")
include("${REPO}/cmake/MapRaycaster.cmake")
include(CTest)
find_path(GLM_INCLUDE_DIR glm/vec3.hpp HINTS /deps ENV GLM_INCLUDE REQUIRED)
find_package(ZLIB REQUIRED)
add_executable(geometry_test geometry_test.cpp "${REPO}/zone/map_geometry.cpp" "${REPO}/common/compression.cpp")
target_include_directories(geometry_test PRIVATE "${REPO}" "${GLM_INCLUDE_DIR}")
target_link_libraries(geometry_test PRIVATE ZLIB::ZLIB)
add_test(NAME geometry COMMAND geometry_test)
add_executable(reload_ownership "${REPO}/tests/map_reload/ownership.cpp")
target_include_directories(reload_ownership PRIVATE "${REPO}")
add_test(NAME reload_ownership COMMAND reload_ownership)
add_executable(observation_test observation_test.cpp)
target_include_directories(observation_test PRIVATE "${REPO}" "${GLM_INCLUDE_DIR}")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME observation_controls COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/observation_controls.py" "$<TARGET_FILE:observation_test>" "ON")
set(check_targets geometry_test reload_ownership observation_test)
find_path(FMT_INCLUDE_DIR fmt/format.h HINTS /deps ENV FMT_INCLUDE REQUIRED)
add_executable(bridge_test bridge_test.cpp "${REPO}/zone/rust_raycast_mesh.cpp")
target_include_directories(bridge_test PRIVATE "${REPO}/zone")
eqemu_link_map_raycaster(bridge_test)
add_test(NAME bridge COMMAND bridge_test)
add_executable(wrapper_test wrapper_test.cpp "${REPO}/zone/map.cpp" "${REPO}/zone/map_geometry.cpp" "${REPO}/zone/rust_raycast_mesh.cpp" "${REPO}/common/compression.cpp")
target_include_directories(wrapper_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/shim" "${REPO}" "${GLM_INCLUDE_DIR}" "${FMT_INCLUDE_DIR}")
target_compile_definitions(wrapper_test PRIVATE FMT_HEADER_ONLY)
target_link_libraries(wrapper_test PRIVATE ZLIB::ZLIB)
eqemu_link_map_raycaster(wrapper_test)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/wrapper_test-data")
add_test(NAME wrapper_test COMMAND wrapper_test)
set_tests_properties(wrapper_test PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/wrapper_test-data")
list(APPEND check_targets bridge_test wrapper_test)
# These tests use assertions in every build configuration.
foreach(check IN LISTS check_targets)
    if(MSVC)
        target_compile_options(${check} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${check} PRIVATE -UNDEBUG)
    endif()
endforeach()
